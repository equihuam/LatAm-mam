read_one <- function(path) {
  # Example filename: "T4.1.web.mix_v1.0.json"
  base <- basename(path)
  m <- str_match(base, "^([A-Z]\\d+)\\.(\\d+)")  # e.g., T4 . 1
  if (is.na(m[1,1])) return(NULL)
  biome_code <- m[1,2]                  # "T4"
  efg_code   <- paste0(m[1,2], ".", m[1,3])  # "T4.1"
  realm      <- substr(efg_code, 1, 1)  # "T" / "F" / "M" / etc.

  suppressWarnings({
    g <- tryCatch(st_read(path, quiet = TRUE), error = function(e) NULL)
  })
  if (is.null(g) || nrow(g) == 0) return(NULL)

  g <- st_transform(g, 3857) |> st_make_valid()

  # Occurrence: normalize available field (occurrence/value/class) -> major/minor
  occ_field <- intersect(c("occurrence","value","class","VAL","val"), names(g))
  if (length(occ_field)) {
    occ <- g[[occ_field[1]]]
    occ <- dplyr::recode(as.character(occ), "1"="major","2"="minor", .default="major")
  } else {
    occ <- "major"
  }

  g |>
    mutate(
      realm = realm,
      biome_code = biome_code,
      efg_code = efg_code,
      efg_name = dplyr::coalesce(efg_lookup$efg_name[match(efg_code, efg_lookup$efg_code)], efg_code),
      occurrence = occ
    ) |>
    select(realm, biome_code, efg_code, efg_name, occurrence, geometry)
}
